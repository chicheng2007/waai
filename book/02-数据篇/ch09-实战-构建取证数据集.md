## 9.2 端到端脚本

把前面各章的代码整合起来。为了可读性，我把它拆成几个函数：

```python
# build_dataset.py
"""星海案：端到端构建取证指令数据集"""
import json
import random
import re
from collections import defaultdict
from datetime import datetime, timedelta

random.seed(42)

# ---------- 第1步：合成原始日志 ----------
def generate_raw_logs():
    normal_ips = [f"10.0.1.{i}" for i in range(2, 50)]
    external_ips = [f"203.0.113.{i}" for i in range(1, 30)]
    base = datetime(2026, 10, 15, 0, 0, 0)
    records = []

    def rec(t, action, src, dst, port, size, etype):
        return {
            "time": t.strftime("%Y-%m-%d %H:%M:%S"),
            "action": action, "src": src, "dst": dst,
            "port": str(port), "size": size, "etype": etype
        }

    # 正常流量 9500
    for i in range(9500):
        t = base + timedelta(seconds=i * 8)
        records.append(rec(t, "ALLOW", random.choice(normal_ips),
                          "10.0.2.10", random.choice([80,443,53]),
                          random.randint(200, 5000), "normal"))
    # 端口扫描 200
    for i in range(200):
        t = base + timedelta(hours=2, minutes=i)
        records.append(rec(t, "DENY", random.choice(external_ips),
                          "10.0.2.10", random.randint(1,1024), 60, "port_scan"))
    # 登录爆破 100
    for i in range(100):
        t = base + timedelta(hours=5, minutes=i)
        records.append(rec(t, "DENY", random.choice(external_ips),
                          "10.0.2.10", 22, 400, "brute_force"))
    # 数据外传 150
    for i in range(150):
        t = base + timedelta(hours=4, minutes=i)
        records.append(rec(t, "ALLOW", random.choice(normal_ips),
                          random.choice(external_ips), 443,
                          random.randint(50_000_000, 500_000_000), "large_transfer"))
    random.shuffle(records)
    return records

# ---------- 第2步：清洗 ----------
def clean(records):
    # 去重
    seen = set(); out = []
    for r in records:
        key = tuple(r.items())
        if key not in seen:
            seen.add(key); out.append(r)
    # 过滤异常大小（保留标记）
    for r in out:
        r["size_ok"] = 0 <= r["size"] <= 1e12
    return out

# ---------- 第3步：增强少数类 ----------
def augment(records):
    from collections import Counter
    counts = Counter(r["etype"] for r in records)
    target = counts["normal"] // 5  # 每类补到正常的 1/5
    augmented = list(records)
    for etype, cnt in counts.items():
        if etype == "normal" or cnt >= target:
            continue
        pool = [r for r in records if r["etype"] == etype]
        need = target - cnt
        for _ in range(need):
            r = random.choice(pool).copy()
            r["size"] = int(r["size"] * random.uniform(0.9, 1.1))
            augmented.append(r)
    return augmented

# ---------- 第4步：指令化 ----------
INSTR = [
    "分析这段日志，判断是否有异常",
    "这段日志正常吗？如果有问题是什么问题",
    "请识别下面日志的事件类型",
    "作为取证分析师，判断这段日志的风险",
    "请用一句话总结下面日志的活动",
]
ANSWER = {
    "normal": "结论：正常流量。依据：内网地址间通信，端口与流量均在正常范围。",
    "port_scan": "结论：端口扫描。依据：单一来源短时间访问大量端口。建议：封禁来源 IP。",
    "brute_force": "结论：登录爆破。依据：同一来源多次登录失败。建议：检查账户安全。",
    "large_transfer": "结论：疑似数据外传。依据：超大流量指向外部地址。建议：核查传输内容。",
}

def to_instruction(records):
    out = []
    for i, r in enumerate(records):
        log = f"{r['time']} {r['action']} {r['src']}->{r['dst']}:{r['port']} size={r['size']}"
        out.append({"messages": [
            {"role": "system", "content": "你是专业的数字取证分析助手。"},
            {"role": "user", "content": f"{INSTR[i % len(INSTR)]}：\n{log}"},
            {"role": "assistant", "content": ANSWER[r["etype"]]}
        ]})
    return out

# ---------- 第5步：划分 + 保存 ----------
def main():
    raw = generate_raw_logs()
    print(f"原始日志: {len(raw)}")
    cleaned = clean(raw)
    print(f"清洗后: {len(cleaned)}")
    augmented = augment(cleaned)
    print(f"增强后: {len(augmented)}")
    samples = to_instruction(augmented)
    random.shuffle(samples)

    split = int(len(samples) * 0.9)
    train, val = samples[:split], samples[split:]

    with open("starsea_sft.jsonl", "w", encoding="utf-8") as f:
        for s in train:
            f.write(json.dumps(s, ensure_ascii=False) + "\n")
    with open("starsea_val.jsonl", "w", encoding="utf-8") as f:
        for s in val:
            f.write(json.dumps(s, ensure_ascii=False) + "\n")
    print(f"训练集 {len(train)}，验证集 {len(val)}")

if __name__ == "__main__":
    main()
```

运行：

```bash
python build_dataset.py
```

你会看到类似输出：

```
原始日志: 9950
清洗后: 9850
增强后: 18250
训练集 16425，验证集 1825
```

## 9.3 数据说明文档

建一个 `DATA.md`，记录这份数据的来龙去脉：

```markdown
# 星海案数据集说明

- 版本：v1.0
- 生成日期：2026-11-27
- 生成脚本：build_dataset.py
- 随机种子：42（可复现）

## 数据来源
全部为合成数据，使用文档保留 IP 段（10.x.x.x / 203.0.113.x）。

## 事件类型分布
- normal: 9500
- port_scan: 200 → 增强至 1900
- brute_force: 100 → 增强至 1900
- large_transfer: 150 → 增强至 1900

## 用途
- starsea_sft.jsonl：SFT 训练集
- starsea_val.jsonl：验证集

## 合法性
纯合成数据，不含真实用户信息，无隐私风险。
```

这份文档在后面调试模型时非常有用——你能清楚知道模型吃的什么数据。

## 9.4 质量验收

生成后做最后验收：

```python
import json
from collections import Counter

samples = [json.loads(l) for l in open("starsea_sft.jsonl")]
print("总条数:", len(samples))

# 检查格式
assert all("messages" in s for s in samples)
assert all(len(s["messages"]) == 3 for s in samples)

# 检查回答分布
answers = Counter(s["messages"][2]["content"] for s in samples)
for a, c in answers.items():
    print(c, a[:20])

# 随机看几条
for s in random.sample(samples, 3):
    print("---")
    print(s["messages"][1]["content"][:80])
    print(s["messages"][2]["content"])
```

确认：格式正确、回答分布均衡、内容合理。通过后，这份数据就可以拿去训练了。
