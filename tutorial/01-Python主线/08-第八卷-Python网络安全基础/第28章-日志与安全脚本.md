## 28.1 常见日志格式

### 28.1.1 Nginx 访问日志

```
192.168.1.1 - - [25/Nov/2026:10:00:00 +0800] "GET /index.html HTTP/1.1" 200 1234 "-" "Mozilla/5.0"
```

字段：IP、时间、请求方法、路径、协议、状态码、响应大小、Referer、User-Agent。

### 28.1.2 Syslog

```
Nov 25 10:00:00 server1 sshd[1234]: Failed password for root from 10.0.0.1 port 22
```

### 28.1.3 Windows 事件日志

通常导出为 CSV 或 JSON 格式，包含事件 ID、时间、来源、描述等。

## 28.2 用正则表达式解析日志

```python
import re

log_line = '192.168.1.1 - - [25/Nov/2026:10:00:00 +0800] "GET /index.html HTTP/1.1" 200 1234 "-" "Mozilla/5.0"'

# 正则解析
pattern = r'(\S+) - - \[(.*?)\] "(\S+) (\S+) (\S+)" (\d+) (\d+)'
match = re.match(pattern, log_line)

if match:
    ip, timestamp, method, path, protocol, status, size = match.groups()
    print(f"IP: {ip}")
    print(f"时间: {timestamp}")
    print(f"请求: {method} {path}")
    print(f"状态码: {status}")
```

## 28.3 实用安全脚本

### 28.3.1 暴力破解检测

```python
import re
from collections import defaultdict
from datetime import datetime

def detect_bruteforce(log_file, threshold=5, window_minutes=10):
    """从 SSH 日志中检测暴力破解"""
    failed_attempts = defaultdict(list)
    
    with open(log_file, "r", encoding="utf-8", errors="ignore") as f:
        for line in f:
            if "Failed password" in line:
                # 提取 IP
                ip_match = re.search(r'from (\d+\.\d+\.\d+\.\d+)', line)
                if ip_match:
                    ip = ip_match.group(1)
                    failed_attempts[ip].append(datetime.now())  # 简化：用当前时间
    
    # 找出超过阈值的 IP
    suspects = {ip: len(times) for ip, times in failed_attempts.items() if len(times) >= threshold}
    return suspects
```

### 28.3.2 Web 攻击特征检测

```python
# 常见 Web 攻击特征
ATTACK_PATTERNS = {
    "SQL注入": re.compile(r"(union.*select|'.*or.*'|--|/\*)", re.I),
    "XSS": re.compile(r"(<script|javascript:|onerror=|onload=)", re.I),
    "目录遍历": re.compile(r"(\.\./|\.\\.\\)"),
    "命令注入": re.compile(r"(;|&&|\|).*(cat|ls|whoami|cmd|powershell)", re.I),
}

def detect_web_attack(request_path):
    """检测请求路径中的攻击特征"""
    detected = []
    for attack_type, pattern in ATTACK_PATTERNS.items():
        if pattern.search(request_path):
            detected.append(attack_type)
    return detected

# 测试
test_requests = [
    "/index.html",
    "/admin?id=1' OR '1'='1",
    "/../../etc/passwd",
    "/search?q=<script>alert(1)</script>",
]

for req in test_requests:
    attacks = detect_web_attack(req)
    print(f"{req}: {attacks if attacks else '正常'}")
```

### 28.3.3 简单的日志告警系统

```python
import time
from collections import deque

class AlertSystem:
    """简单的日志告警系统"""
    
    def __init__(self, threshold=10, window=60):
        self.threshold = threshold
        self.window = window
        self.events = deque()
    
    def add_event(self, event_type, source):
        """添加事件，检查是否触发告警"""
        now = time.time()
        self.events.append((now, event_type, source))
        
        # 清理过期事件
        while self.events and now - self.events[0][0] > self.window:
            self.events.popleft()
        
        # 统计该来源的事件数
        count = sum(1 for _, t, s in self.events if s == source and t == event_type)
        
        if count >= self.threshold:
            return f"[告警] {source} 在 {self.window} 秒内触发 {count} 次 {event_type}"
        return None

# 使用
alert = AlertSystem(threshold=3, window=60)
for i in range(5):
    result = alert.add_event("LOGIN_FAILED", "10.0.0.1")
    if result:
        print(result)
```
