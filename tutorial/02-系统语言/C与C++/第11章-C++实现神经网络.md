## 11.1 矩阵类：AI 的基石

```cpp
#include <vector>
#include <iostream>
#include <cassert>

class Matrix {
private:
    size_t rows, cols;
    std::vector<double> data;   // 连续存储（缓存友好）

public:
    Matrix(size_t r, size_t c, double val = 0.0) 
        : rows(r), cols(c), data(r * c, val) {}
    
    double& operator()(size_t i, size_t j) { return data[i * cols + j]; }
    double operator()(size_t i, size_t j) const { return data[i * cols + j]; }
    
    size_t getRows() const { return rows; }
    size_t getCols() const { return cols; }
    
    // 矩阵乘法
    Matrix operator*(const Matrix& other) const {
        assert(cols == other.rows);
        Matrix result(rows, other.cols);
        for (size_t i = 0; i < rows; i++) {
            for (size_t k = 0; k < cols; k++) {
                for (size_t j = 0; j < other.cols; j++) {
                    result(i, j) += (*this)(i, k) * other(k, j);
                }
            }
        }
        return result;
    }
    
    // 矩阵加法
    Matrix operator+(const Matrix& other) const {
        assert(rows == other.rows && cols == other.cols);
        Matrix result(rows, cols);
        for (size_t i = 0; i < data.size(); i++) {
            result.data[i] = data[i] + other.data[i];
        }
        return result;
    }
    
    void print() const {
        for (size_t i = 0; i < rows; i++) {
            for (size_t j = 0; j < cols; j++) {
                std::cout << (*this)(i, j) << " ";
            }
            std::cout << std::endl;
        }
    }
};
```

**关键点**：用一维 `vector` 存储二维矩阵（`data[i*cols+j]`），这是连续内存，缓存友好。高性能计算都这么做。

## 11.2 激活函数

```cpp
#include <cmath>

// ReLU
double relu(double x) { return x > 0 ? x : 0; }
double relu_deriv(double x) { return x > 0 ? 1 : 0; }

// Sigmoid
double sigmoid(double x) { return 1.0 / (1.0 + std::exp(-x)); }
double sigmoid_deriv(double x) {
    double s = sigmoid(x);
    return s * (1 - s);
}

// 对矩阵每个元素应用函数
Matrix apply(const Matrix& m, double (*func)(double)) {
    Matrix result(m.getRows(), m.getCols());
    for (size_t i = 0; i < m.getRows(); i++) {
        for (size_t j = 0; j < m.getCols(); j++) {
            result(i, j) = func(m(i, j));
        }
    }
    return result;
}
```

## 11.3 两层神经网络

```cpp
class NeuralNetwork {
private:
    Matrix W1, b1, W2, b2;
    double learning_rate;
    
public:
    NeuralNetwork(size_t input_size, size_t hidden_size, size_t output_size, double lr)
        : W1(input_size, hidden_size), b1(1, hidden_size),
          W2(hidden_size, output_size), b2(1, output_size),
          learning_rate(lr) {
        // 随机初始化权重（简单示例，实际用 Xavier/He 初始化）
        srand(time(nullptr));
        for (size_t i = 0; i < input_size; i++)
            for (size_t j = 0; j < hidden_size; j++)
                W1(i, j) = ((double)rand() / RAND_MAX - 0.5) * 0.5;
        for (size_t i = 0; i < hidden_size; i++)
            for (size_t j = 0; j < output_size; j++)
                W2(i, j) = ((double)rand() / RAND_MAX - 0.5) * 0.5;
    }
    
    // 前向传播
    Matrix forward(const Matrix& X) {
        Matrix z1 = X * W1 + b1;
        Matrix a1 = apply(z1, relu);
        Matrix z2 = a1 * W2 + b2;
        Matrix a2 = apply(z2, sigmoid);
        return a2;
    }
    
    // 训练（简化版，单样本）
    void train(const Matrix& X, const Matrix& y) {
        // === 前向传播 ===
        Matrix z1 = X * W1 + b1;
        Matrix a1 = apply(z1, relu);
        Matrix z2 = a1 * W2 + b2;
        Matrix a2 = apply(z2, sigmoid);
        
        // === 反向传播 ===
        // 输出层误差: δ2 = (a2 - y) * sigmoid'(z2)
        Matrix error = a2 + (y * -1.0);   // a2 - y
        Matrix delta2(error.getRows(), error.getCols());
        for (size_t i = 0; i < error.getRows(); i++)
            for (size_t j = 0; j < error.getCols(); j++)
                delta2(i, j) = error(i, j) * sigmoid_deriv(z2(i, j));
        
        // 隐藏层误差: δ1 = δ2 * W2^T * relu'(z1)
        Matrix W2_T(W2.getCols(), W2.getRows());
        for (size_t i = 0; i < W2.getRows(); i++)
            for (size_t j = 0; j < W2.getCols(); j++)
                W2_T(j, i) = W2(i, j);
        
        Matrix delta1 = delta2 * W2_T;
        for (size_t i = 0; i < delta1.getRows(); i++)
            for (size_t j = 0; j < delta1.getCols(); j++)
                delta1(i, j) *= relu_deriv(z1(i, j));
        
        // === 更新权重 ===
        // W2 -= lr * a1^T * δ2
        Matrix a1_T(a1.getCols(), a1.getRows());
        for (size_t i = 0; i < a1.getRows(); i++)
            for (size_t j = 0; j < a1.getCols(); j++)
                a1_T(j, i) = a1(i, j);
        
        Matrix gradW2 = a1_T * delta2;
        for (size_t i = 0; i < W2.getRows(); i++)
            for (size_t j = 0; j < W2.getCols(); j++)
                W2(i, j) -= learning_rate * gradW2(i, j);
        
        // W1 -= lr * X^T * δ1
        Matrix X_T(X.getCols(), X.getRows());
        for (size_t i = 0; i < X.getRows(); i++)
            for (size_t j = 0; j < X.getCols(); j++)
                X_T(j, i) = X(i, j);
        
        Matrix gradW1 = X_T * delta1;
        for (size_t i = 0; i < W1.getRows(); i++)
            for (size_t j = 0; j < W1.getCols(); j++)
                W1(i, j) -= learning_rate * gradW1(i, j);
    }
};
```

> 这段代码展示了反向传播的**完整数学过程**。PyTorch 的 `autograd` 帮你自动做了这些事，但原理就是这样。

## 11.4 训练示例：学习 XOR

```cpp
int main() {
    NeuralNetwork nn(2, 4, 1, 0.5);
    
    // XOR 训练数据
    Matrix X(4, 2);
    X(0,0)=0; X(0,1)=0;
    X(1,0)=0; X(1,1)=1;
    X(2,0)=1; X(2,1)=0;
    X(3,0)=1; X(3,1)=1;
    
    Matrix y(4, 1);
    y(0,0)=0; y(1,0)=1; y(2,0)=1; y(3,0)=0;
    
    // 训练
    for (int epoch = 0; epoch < 10000; epoch++) {
        nn.train(X, y);
    }
    
    // 测试
    Matrix result = nn.forward(X);
    std::cout << "XOR 预测结果:\n";
    result.print();
    
    return 0;
}
```

输出接近：

```
0.03
0.97
0.96
0.04
```

网络学会了 XOR！这是深度学习最经典的入门例子。

## 11.5 性能优化的启示

### 11.5.1 为什么手写慢

上面的实现用了三层嵌套循环做矩阵乘法，有大量函数调用和临时对象。真实框架做了这些优化：

1. **向量化**：用 SIMD 指令并行计算（一次算多个数）
2. **BLAS 库**：调用高度优化的矩阵运算库（OpenBLAS、MKL）
3. **GPU 并行**：CUDA 核心几千个线程同时算
4. **内存布局优化**：连续存储、缓存预取

### 11.5.2 性能对比感受


## 11.7 用 Eigen 库重写（工程化做法）

手写矩阵类用于学习很好，但真实项目应该用成熟的线性代数库。**Eigen** 是 C++ 最流行的头文件库：

```cpp
#include <Eigen/Dense>
#include <iostream>

using namespace Eigen;

int main() {
    // 定义矩阵（编译时知道大小，零开销）
    Matrix2d a;
    a << 1, 2, 3, 4;
    
    Matrix2d b;
    b << 5, 6, 7, 8;
    
    // 矩阵乘法（一行代码，内部高度优化）
    Matrix2d c = a * b;
    std::cout << c << std::endl;
    
    // 动态大小矩阵
    MatrixXd dyn(3, 3);
    dyn = MatrixXd::Random(3, 3);
    
    // 常用运算
    std::cout << "转置:\n" << dyn.transpose() << std::endl;
    std::cout << "逆:\n" << dyn.inverse() << std::endl;
    std::cout << "行列式: " << dyn.determinant() << std::endl;
    
    // 求解线性方程组 Ax = b
    VectorXd bvec(3);
    bvec << 1, 2, 3;
    VectorXd x = dyn.colPivHouseholderQr().solve(bvec);
    
    return 0;
}
```

编译（Eigen 是头文件库，无需链接）：

```bash
g++ -std=c++17 -O2 -I/usr/include/eigen3 eigen_demo.cpp -o eigen_demo
```

**Eigen 的优势**：

1. **表达式模板**：`a * b + c` 不会创建临时对象，编译后是最优的循环
2. **向量化**：自动用 SIMD 指令
3. **零开销抽象**：高层 API 和手写优化代码性能相当
4. **类型安全**：编译期检查维度匹配

> **工程启示**：学习阶段手写理解原理，生产阶段用成熟库。这体现了"知道底层"和"善用工具"的平衡——这正是 C++ 工程师的价值所在。

## 11.8 从零实现 vs 框架：深度理解的价值

用心写完上面的神经网络后，你会对以下概念有"肌肉记忆"级的深刻理解：

| 概念 | 手写后的理解 |
|------|------------|
| 矩阵乘法 | 明白了为什么顺序影响性能（缓存局部性） |
| 反向传播 | 不再觉得是魔法，就是链式法则的代码化 |
| 梯度下降 | 理解了学习率、梯度的实际含义 |
| 权重初始化 | 知道了随机初始化的重要性 |
| 过拟合 | 能想到为什么需要正则化、dropout |

这些理解让你在调 PyTorch 模型时不再盲目：

- 损失不下降？你知道可能是学习率太大、梯度爆炸、初始化不当
- 内存爆了？你知道是 batch 太大、中间变量没释放
- 速度慢？你知道是数据加载、矩阵运算还是内存布局的问题

> **这就是"懂底层"的回报**。很多 AI 工程师只会调 API，出了问题束手无策。而你因为懂底层，能真正诊断和优化。

## 11.9 性能优化的更多技巧

```cpp
// 1. 循环展开（手动或编译器自动）
// 2. 分块（tiling）：把大矩阵分成能装进 CPU 缓存的小块
// 3. 避免虚函数调用（内联）
// 4. 用 Eigen 的 noalias() 避免临时对象
c.noalias() = a * b;   // 告诉 Eigen c 和 a/b 不重叠，省去临时拷贝

// 5. 多线程（OpenMP）
#pragma omp parallel for
for (int i = 0; i < n; i++) {
    // 并行计算
}
```
