## 33.2 完整数据库设计

一个整合的数据库，包含所有核心表：

```sql
CREATE DATABASE IF NOT EXISTS aisecurity DEFAULT CHARSET utf8mb4;
USE aisecurity;

-- 1. 用户表
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(64) NOT NULL UNIQUE,
    password_hash CHAR(64) NOT NULL COMMENT 'SHA256哈希，不存明文',
    role ENUM('ADMIN','ANALYST','VIEWER') NOT NULL DEFAULT 'VIEWER',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 2. 事件表（统一的安全事件）
CREATE TABLE security_events (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    event_type ENUM('LOGIN_FAIL','PORT_SCAN','SQL_INJECT','MALWARE','ANOMALY') NOT NULL,
    src_ip VARCHAR(64) NOT NULL,
    username VARCHAR(64),
    severity TINYINT UNSIGNED NOT NULL DEFAULT 3,
    features JSON COMMENT '事件特征（供AI使用）',
    raw_data JSON COMMENT '原始数据',
    is_malicious BOOLEAN COMMENT 'AI判定：是否恶意',
    risk_score DECIMAL(5,4) COMMENT 'AI风险评分',
    occurred_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_type_time (event_type, occurred_at),
    INDEX idx_ip (src_ip),
    INDEX idx_risk (risk_score)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 3. 模型表
CREATE TABLE models (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(128) NOT NULL,
    version VARCHAR(32) NOT NULL,
    metrics JSON,
    is_active BOOLEAN DEFAULT FALSE,
    deployed_at DATETIME,
    UNIQUE KEY uk_name_version (name, version)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 4. 告警表
CREATE TABLE alerts (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    event_id BIGINT UNSIGNED NOT NULL,
    alert_type VARCHAR(64) NOT NULL,
    severity TINYINT UNSIGNED NOT NULL,
    risk_score DECIMAL(5,4),
    status ENUM('OPEN','INVESTIGATING','CLOSED') DEFAULT 'OPEN',
    assignee INT COMMENT '处理人',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_status (status),
    INDEX idx_severity (severity),
    FOREIGN KEY (event_id) REFERENCES security_events(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 5. 审计表
CREATE TABLE audit_log (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    action VARCHAR(64) NOT NULL,
    target VARCHAR(255),
    detail JSON,
    ip VARCHAR(64),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_user_time (user_id, created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

设计亮点：

- 统一的事件表，用 ENUM 区分类型，JSON 存灵活特征
- `risk_score` 字段存 AI 评分，可建索引按风险查询
- 外键保证告警关联的事件存在
- 审计表 append-only（权限控制）
- 用户信息密码存哈希，不存明文

## 33.3 数据管道：从日志到特征

用 Python 实现数据入库和特征提取：

```python
import pymysql
import hashlib
import json
from datetime import datetime

DB_CONFIG = {
    'host': 'localhost', 'user': 'secapp',
    'password': '从环境变量读', 'database': 'aisecurity',
    'charset': 'utf8mb4'
}

def insert_event(event_type, src_ip, severity, raw_data):
    """插入安全事件，并提取特征"""
    features = extract_features(event_type, src_ip, raw_data)

    conn = pymysql.connect(**DB_CONFIG)
    try:
        with conn.cursor() as cur:
            cur.execute("""
                INSERT INTO security_events
                (event_type, src_ip, severity, features, raw_data, occurred_at)
                VALUES (%s, %s, %s, %s, %s, %s)
            """, (event_type, src_ip, severity,
                  json.dumps(features), json.dumps(raw_data), datetime.now()))
            event_id = cur.lastrowid
        conn.commit()
        return event_id
    finally:
        conn.close()

def extract_features(event_type, src_ip, raw_data):
    """从原始数据提取特征（供AI使用）"""
    features = {
        'ip_reputation': check_ip_reputation(src_ip),  # 查情报库
        'hour_of_day': datetime.now().hour,
        'data_length': len(json.dumps(raw_data)),
    }
    if event_type == 'LOGIN_FAIL':
        features['fail_count_1h'] = count_recent_fails(src_ip, 1)
    return features

def check_ip_reputation(ip):
    """查威胁情报（简化）"""
    conn = pymysql.connect(**DB_CONFIG)
    try:
        with conn.cursor() as cur:
            cur.execute("""
                SELECT MAX(severity) FROM threat_intel.indicators
                WHERE indicator=%s AND type='IP' AND is_active=TRUE
            """, (ip,))
            row = cur.fetchone()
            return row[0] or 0
    finally:
        conn.close()

def count_recent_fails(ip, hours):
    """统计最近N小时失败次数"""
    conn = pymysql.connect(**DB_CONFIG)
    try:
        with conn.cursor() as cur:
            cur.execute("""
                SELECT COUNT(*) FROM security_events
                WHERE src_ip=%s AND event_type='LOGIN_FAIL'
                  AND occurred_at >= NOW() - INTERVAL %s HOUR
            """, (ip, hours))
            return cur.fetchone()[0]
    finally:
        conn.close()
```

这段代码展示了数据管道：事件入库 → 特征工程（查情报、统计历史）→ 特征 JSON 存回数据库，供 AI 模型使用。

## 33.4 AI 检测层

用训练好的模型对事件评分（简化示例）：

```python
import joblib
import pymysql

# 加载模型（模型文件由训练脚本生成，元数据在 models 表）
model = joblib.load('/models/risk_detector.pkl')

def detect_and_alert():
    """读取未判定的事件，用AI检测，生成告警"""
    conn = pymysql.connect(**DB_CONFIG)
    try:
        with conn.cursor(pymysql.cursors.DictCursor) as cur:
            cur.execute("""
                SELECT id, event_type, features, severity
                FROM security_events
                WHERE is_malicious IS NULL
                LIMIT 100
            """)
            events = cur.fetchall()

        for event in events:
            features = json.loads(event['features'])
            # 构造特征向量
            vector = [[
                features.get('ip_reputation', 0),
                features.get('hour_of_day', 12),
                features.get('fail_count_1h', 0),
                features.get('data_length', 0),
            ]]
            # 预测
            risk = model.predict_proba(vector)[0][1]  # 恶意概率
            is_malicious = risk > 0.5

            # 回写结果 + 生成告警
            with conn.cursor() as cur:
                cur.execute("""
                    UPDATE security_events
                    SET is_malicious=%s, risk_score=%s
                    WHERE id=%s
                """, (is_malicious, round(risk, 4), event['id']))

                if is_malicious:
                    cur.execute("""
                        INSERT INTO alerts (event_id, alert_type, severity, risk_score)
                        VALUES (%s, %s, %s, %s)
                    """, (event['id'], event['event_type'], event['severity'], round(risk, 4)))
        conn.commit()
        print(f"检测了 {len(events)} 条事件")
    finally:
        conn.close()
```

AI 检测闭环：读未判定事件 → 模型预测风险 → 回写结果 → 高危生成告警。这就是 AI + 安全的核心逻辑。

## 33.5 服务层：FastAPI 接口

```python
from fastapi import FastAPI, HTTPException
import pymysql

app = FastAPI(title="AI Security Backend")

def query(sql, params=None):
    conn = pymysql.connect(**DB_CONFIG)
    try:
        with conn.cursor(pymysql.cursors.DictCursor) as cur:
            cur.execute(sql, params or ())
            return cur.fetchall()
    finally:
        conn.close()

@app.get("/health")
def health():
    return {"status": "ok"}

@app.get("/alerts")
def get_alerts(status: str = "OPEN", limit: int = 50):
    """查询告警（参数化查询，防注入）"""
    return query("""
        SELECT a.id, a.alert_type, a.severity, a.risk_score, a.created_at,
               e.src_ip, e.event_type
        FROM alerts a
        JOIN security_events e ON a.event_id = e.id
        WHERE a.status = %s
        ORDER BY a.severity DESC, a.created_at DESC
        LIMIT %s
    """, (status, limit))

@app.get("/stats")
def get_stats():
    """统计报表"""
    return query("""
        SELECT event_type,
               COUNT(*) AS total,
               SUM(is_malicious) AS malicious,
               AVG(risk_score) AS avg_risk
        FROM security_events
        WHERE occurred_at >= NOW() - INTERVAL 1 DAY
        GROUP BY event_type
        ORDER BY malicious DESC
    """)

@app.get("/ip/{ip}")
def check_ip(ip: str):
    """查询某IP的风险画像"""
    result = query("""
        SELECT COUNT(*) AS event_count,
               SUM(is_malicious) AS malicious_count,
               MAX(risk_score) AS max_risk
        FROM security_events
        WHERE src_ip = %s
    """, (ip,))
    return result[0]
```

启动：

```bash
uvicorn main:app --reload
```

访问 `http://localhost:8000/docs` 看 API 文档。

**安全要点**：

- 所有查询用参数化（`%s`），防注入
- `/ip/{ip}` 的参数也参数化，即使用户输入注入代码也无效
- 限制 limit 防止一次查太多

## 33.6 安全加固集成

把前面学的安全知识用到项目里：

### 1. 最小权限账号

```sql
-- 应用账号只有必要权限
CREATE USER 'aisec_app'@'localhost' IDENTIFIED BY '强密码';
GRANT SELECT, INSERT, UPDATE ON aisecurity.* TO 'aisec_app'@'localhost';
-- 注意：没有 DELETE、DROP 权限，防止误删/注入破坏
```

### 2. 审计触发器

```sql
DELIMITER $$
CREATE TRIGGER trg_audit_alert
AFTER UPDATE ON alerts
FOR EACH ROW
BEGIN
    INSERT INTO audit_log (action, target, detail, created_at)
    VALUES ('UPDATE_ALERT', NEW.id,
            JSON_OBJECT('old_status', OLD.status, 'new_status', NEW.status),
            NOW());
END$$
DELIMITER ;
```

### 3. 备份策略

```bash
# 每天全量备份 + binlog 增量
mysqldump -u root -p --single-transaction --all-databases \
  --master-data=2 > /backup/aisecurity_$(date +%Y%m%d).sql
```

### 4. 密码存储

```python
# 存哈希，不存明文
password_hash = hashlib.sha256((password + salt).encode()).hexdigest()
```

> **进阶**：生产环境用 bcrypt/argon2（加盐、慢哈希），比 SHA256 更安全。
