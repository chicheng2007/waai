## 31.2 数据库设计

### 表结构

```sql
CREATE DATABASE IF NOT EXISTS soc_lab DEFAULT CHARSET utf8mb4;
USE soc_lab;

-- 1. 登录日志表
CREATE TABLE login_logs (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    ip VARCHAR(64) NOT NULL COMMENT '来源IP',
    username VARCHAR(64) NOT NULL COMMENT '用户名',
    login_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '登录时间',
    result ENUM('SUCCESS','FAILED') NOT NULL COMMENT '结果',
    user_agent VARCHAR(255) COMMENT 'UA',
    INDEX idx_ip_time (ip, login_time),
    INDEX idx_time (login_time),
    INDEX idx_result (result)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='登录日志';

-- 2. 防火墙日志表
CREATE TABLE firewall_logs (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    src_ip VARCHAR(64) NOT NULL COMMENT '源IP',
    dst_ip VARCHAR(64) NOT NULL COMMENT '目的IP',
    dst_port SMALLINT UNSIGNED COMMENT '目的端口',
    protocol ENUM('TCP','UDP','ICMP') NOT NULL,
    action ENUM('ALLOW','DENY') NOT NULL COMMENT '动作',
    event_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_src_time (src_ip, event_time),
    INDEX idx_dst_port (dst_port)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='防火墙日志';

-- 3. 安全告警表
CREATE TABLE security_alerts (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    alert_type ENUM('BRUTE_FORCE','PORT_SCAN','ANOMALY_LOGIN','SQL_INJECT') NOT NULL,
    severity TINYINT UNSIGNED NOT NULL DEFAULT 3 COMMENT '级别1-5',
    ip VARCHAR(64) NOT NULL COMMENT '相关IP',
    detail JSON COMMENT '详情',
    status ENUM('OPEN','INVESTIGATING','CLOSED') DEFAULT 'OPEN',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_type_time (alert_type, created_at),
    INDEX idx_ip (ip)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='安全告警';
```

设计说明：

- 每张表有合适的主键（BIGINT UNSIGNED）
- 常用查询列建索引（ip + time 组合）
- 用 ENUM 约束取值
- JSON 存灵活详情
- 时间戳自动填充

## 31.3 造测试数据

用存储过程批量生成模拟日志：

```sql
DELIMITER $$

CREATE PROCEDURE sp_gen_login_logs(IN n INT)
BEGIN
    DECLARE i INT DEFAULT 0;
    DECLARE rand_ip VARCHAR(64);
    DECLARE rand_user VARCHAR(64);
    DECLARE rand_result ENUM('SUCCESS','FAILED');

    WHILE i < n DO
        -- 随机 IP：模拟几个固定 IP + 随机 IP
        IF RAND() < 0.3 THEN
            SET rand_ip = '45.33.22.11';  -- 模拟攻击者
        ELSE
            SET rand_ip = CONCAT('192.168.1.', FLOOR(RAND()*50+1));
        END IF;

        SET rand_user = ELT(FLOOR(RAND()*4+1), 'admin', 'root', 'test', 'guest');

        -- 45.33.22.11 大部分失败（模拟暴力破解）
        IF rand_ip = '45.33.22.11' THEN
            SET rand_result = IF(RAND() < 0.9, 'FAILED', 'SUCCESS');
        ELSE
            SET rand_result = IF(RAND() < 0.8, 'SUCCESS', 'FAILED');
        END IF;

        INSERT INTO login_logs (ip, username, login_time, result)
        VALUES (rand_ip, rand_user,
                NOW() - INTERVAL FLOOR(RAND()*86400) SECOND,
                rand_result);

        SET i = i + 1;
    END WHILE;
END$$

DELIMITER ;

-- 生成 1000 条登录日志
CALL sp_gen_login_logs(1000);
```

现在你有了一批带"攻击特征"的数据。

## 31.4 分析查询

### 查询 1：找出暴力破解 IP

```sql
SELECT
    ip,
    COUNT(*) AS total,
    SUM(result='FAILED') AS failures,
    COUNT(DISTINCT username) AS users_tried,
    MIN(login_time) AS first_seen,
    MAX(login_time) AS last_seen
FROM login_logs
GROUP BY ip
HAVING failures > 30
ORDER BY failures DESC;
```

这条查询综合了：GROUP BY、SUM 条件计数、COUNT DISTINCT、MIN/MAX、HAVING、ORDER BY。

### 查询 2：检测端口扫描

```sql
-- 找出访问超过 10 个不同端口的源 IP（疑似扫描）
SELECT
    src_ip,
    COUNT(DISTINCT dst_port) AS ports_touched,
    COUNT(*) AS packets,
    SUM(action='DENY') AS denied
FROM firewall_logs
GROUP BY src_ip
HAVING ports_touched > 10
ORDER BY ports_touched DESC;
```

### 查询 3：异常时间登录

```sql
-- 凌晨 0-5 点的登录（异常时段）
SELECT ip, username, login_time, result
FROM login_logs
WHERE HOUR(login_time) BETWEEN 0 AND 5
ORDER BY login_time DESC
LIMIT 50;
```

> **注意**：`HOUR(login_time)` 用了函数，会导致索引失效（第 13、22 章）。数据量大时可以用范围查询改写，或建函数索引（MySQL 8.0 支持）。

### 查询 4：每日统计报表

```sql
SELECT
    DATE(login_time) AS day,
    COUNT(*) AS total_logins,
    SUM(result='FAILED') AS failed_logins,
    COUNT(DISTINCT ip) AS unique_ips,
    ROUND(SUM(result='FAILED')/COUNT(*)*100, 2) AS fail_rate_pct
FROM login_logs
GROUP BY DATE(login_time)
ORDER BY day;
```

## 31.5 用触发器自动生成告警

插入日志时，自动检测暴力破解并生成告警：

```sql
DELIMITER $$

CREATE TRIGGER trg_check_bruteforce
AFTER INSERT ON login_logs
FOR EACH ROW
BEGIN
    DECLARE fail_count INT;

    IF NEW.result = 'FAILED' THEN
        -- 统计该 IP 最近 10 分钟的失败次数
        SELECT COUNT(*) INTO fail_count
        FROM login_logs
        WHERE ip = NEW.ip
          AND result = 'FAILED'
          AND login_time >= NOW() - INTERVAL 10 MINUTE;

        -- 超过 20 次，生成告警（避免重复：检查是否已有 OPEN 告警）
        IF fail_count = 20 AND NOT EXISTS (
            SELECT 1 FROM security_alerts
            WHERE alert_type='BRUTE_FORCE' AND ip=NEW.ip AND status='OPEN'
        ) THEN
            INSERT INTO security_alerts (alert_type, severity, ip, detail)
            VALUES ('BRUTE_FORCE', 4, NEW.ip,
                    JSON_OBJECT('fail_count', fail_count, 'threshold', 20));
        END IF;
    END IF;
END$$

DELIMITER ;
```

`fail_count = 20`（而不是 >20）确保只在达到阈值的那一刻生成一次告警，配合 NOT EXISTS 避免重复。

## 31.6 Python 实现日志入库与检测

```python
import pymysql
import os
from datetime import datetime

DB_CONFIG = {
    'host': 'localhost',
    'user': os.environ.get('DB_USER', 'secapp'),
    'password': os.environ.get('DB_PASSWORD'),
    'database': 'soc_lab',
    'charset': 'utf8mb4'
}

def insert_login(ip, username, result):
    """插入一条登录日志"""
    conn = pymysql.connect(**DB_CONFIG)
    try:
        with conn.cursor() as cur:
            cur.execute(
                """INSERT INTO login_logs (ip, username, login_time, result)
                   VALUES (%s, %s, %s, %s)""",
                (ip, username, datetime.now(), result)
            )
        conn.commit()
    finally:
        conn.close()

def get_open_alerts():
    """查询未处理的告警"""
    conn = pymysql.connect(**DB_CONFIG)
    try:
        with conn.cursor(pymysql.cursors.DictCursor) as cur:
            cur.execute("""
                SELECT id, alert_type, severity, ip, created_at
                FROM security_alerts
                WHERE status='OPEN'
                ORDER BY severity DESC, created_at DESC
            """)
            return cur.fetchall()
    finally:
        conn.close()

def daily_report():
    """生成日报"""
    conn = pymysql.connect(**DB_CONFIG)
    try:
        with conn.cursor(pymysql.cursors.DictCursor) as cur:
            cur.execute("""
                SELECT DATE(login_time) AS day,
                       COUNT(*) AS total,
                       SUM(result='FAILED') AS failed
                FROM login_logs
                GROUP BY DATE(login_time)
                ORDER BY day DESC LIMIT 7
            """)
            return cur.fetchall()
    finally:
        conn.close()

if __name__ == '__main__':
    # 模拟插入日志
    for i in range(30):
        insert_login('45.33.22.11', 'admin', 'FAILED')

    # 查看告警
    print("=== 未处理告警 ===")
    for alert in get_open_alerts():
        print(f"[{alert['severity']}] {alert['alert_type']} - {alert['ip']}")

    # 日报
    print("\n=== 近7日统计 ===")
    for row in daily_report():
        print(f"{row['day']}: 总{row['total']} 失败{row['failed']}")
```

这个脚本演示了：参数化查询入库、查询告警、生成报表——一个简化版 SIEM 的核心逻辑。

## 31.7 索引优化验证

用 EXPLAIN 验证我们的索引设计：

```sql
EXPLAIN SELECT * FROM login_logs
WHERE ip='45.33.22.11' AND login_time >= NOW() - INTERVAL 1 HOUR;
```

应该看到 `type=ref`、`key=idx_ip_time`——索引生效。

```sql
EXPLAIN SELECT ip, COUNT(*) FROM login_logs
WHERE result='FAILED'
GROUP BY ip;
```

应该看到 `key=idx_result`，用上了 result 索引。
